cmake_minimum_required(VERSION 3.21)
project(h2o-clj-shim VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif ()

# Set up parallel build by default using nproc
if (NOT DEFINED ENV{CMAKE_BUILD_PARALLEL_LEVEL})
    cmake_host_system_information(RESULT CPU_CORES QUERY NUMBER_OF_LOGICAL_CORES)
    set(ENV{CMAKE_BUILD_PARALLEL_LEVEL} ${CPU_CORES})
endif()
set(PARALLEL_JOBS $ENV{CMAKE_BUILD_PARALLEL_LEVEL})

if (CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(CMAKE_C_FLAGS_DEBUG "-g3 -O0 -fno-omit-frame-pointer")
endif ()

option(ENABLE_ASAN "Enable AddressSanitizer" OFF)
if (ENABLE_ASAN)
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fsanitize=address -fno-omit-frame-pointer")
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -fsanitize=address")
    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -fsanitize=address")
endif ()

set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -DH2O_USE_LIBUV=0")

if (CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -DCLJ_DEBUG=1")
endif ()
set(CMAKE_C_FLAGS_RELEASE "-O2 -g")

if(CMAKE_SYSTEM_NAME MATCHES "Darwin")
    add_definitions(-D_DARWIN_C_SOURCE)
endif()

if(CMAKE_SYSTEM_NAME MATCHES "Linux")
    add_definitions(-D_GNU_SOURCE)
endif()

add_definitions(-D_POSIX_C_SOURCE=200809L)

find_package(PkgConfig REQUIRED)
pkg_check_modules(H2O REQUIRED libh2o-evloop)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)


# Handle math library (libm) - on some systems it's built into libc
include(CheckLibraryExists)
check_library_exists(m sin "" HAVE_LIBM)
if (HAVE_LIBM)
    set(MATH_LIBRARY m)
else ()
    set(MATH_LIBRARY "")
endif ()

set(SRC_DIR src/main/native)
include_directories(${SRC_DIR})
include_directories(${H2O_INCLUDE_DIRS})
add_definitions(${H2O_CFLAGS_OTHER})
link_directories(${H2O_LIBRARY_DIRS})

set(SOURCES
    ${SRC_DIR}/shim.c
)
add_library(h2oclj SHARED ${SOURCES})
set_target_properties(h2oclj PROPERTIES
        VERSION ${PROJECT_VERSION}
        SOVERSION ${PROJECT_VERSION_MAJOR}
        OUTPUT_NAME h2oclj
)

# Link libraries to the shared library
target_link_libraries(h2oclj
        ${H2O_LIBRARIES}
        OpenSSL::SSL
        OpenSSL::Crypto
        ${MATH_LIBRARY}
        Threads::Threads
        ZLIB::ZLIB
)


# Generate compile_flags.txt for LSP (equivalent to Makefile target)
add_custom_target(compile_flags
        COMMAND ${CMAKE_COMMAND} -E echo "# Auto-generated by CMake - do not edit" > compile_flags.txt
        COMMAND ${CMAKE_COMMAND} -E echo "${CMAKE_C_FLAGS}" | tr ' ' '\\n' | grep -v '^$$' >> compile_flags.txt || true
        COMMAND ${CMAKE_COMMAND} -E echo "-I${CMAKE_CURRENT_SOURCE_DIR}/${SRC_DIR}" >> compile_flags.txt
        COMMAND pkg-config --cflags libh2o-evloop | tr ' ' '\\n' | grep -v '^$$' >> compile_flags.txt || true
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        COMMENT "Generating compile_flags.txt for LSP"
)

# Lint targets
find_program(CLANG_TIDY_EXECUTABLE clang-tidy)
if (CLANG_TIDY_EXECUTABLE)
    add_custom_target(lint
            COMMAND ${CLANG_TIDY_EXECUTABLE} ${SOURCES} -- ${CMAKE_C_FLAGS} ${H2O_CFLAGS} -I${CMAKE_CURRENT_SOURCE_DIR}/${SRC_DIR}
            WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
            COMMENT "Running clang-tidy on C source files"
    )
endif ()

# Installation targets
install(TARGETS h2oclj
        LIBRARY DESTINATION lib
        ARCHIVE DESTINATION lib
        RUNTIME DESTINATION bin
)

install(FILES ${SRC_DIR}/shim.h
        DESTINATION include
)

# Print configuration summary
message(STATUS "Build configuration:")
message(STATUS "  Build Type: ${CMAKE_BUILD_TYPE}")
message(STATUS "  C Compiler: ${CMAKE_C_COMPILER}")
message(STATUS "  C Flags: ${CMAKE_C_FLAGS}")
message(STATUS "  Parallel Jobs: ${PARALLEL_JOBS}")
message(STATUS "  H2O Found: ${H2O_FOUND}")
message(STATUS "  H2O Include Dirs: ${H2O_INCLUDE_DIRS}")
message(STATUS "  H2O Libraries: ${H2O_LIBRARIES}")
message(STATUS "  OpenSSL Found: ${OPENSSL_FOUND}")
message(STATUS "  Enable ASan: ${ENABLE_ASAN}")
